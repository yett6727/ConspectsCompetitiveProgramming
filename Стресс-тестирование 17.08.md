```cpp
#include <sstream>

string generate(){
	oss << dist(gen);
}

void factorize(isstream& iss){
	vector<int>ret;
	int x; iss >> x;
	if (x == 1) return;
	
	for (int k = 2; k * k <= x; k++){
		if (!(x % k)) ret.push_back(k);
		while (!(x % k)) x /= k;
	}
	
	if (x != 1) ret.push_back(x);
	for (auto elem : ret) oss << elem << ' ';
}

void stupid(isstream& iss){
	int x; iss >> x;
	//тупое решение лень писать
}

void stress(){
	while (true){
		string test = generate();
		istringstream iss1(test), iss2(test);

			cerr << "test failed: " << test << ' ' << a << ' ' << b;
			exit(-1);
		} else {
			cerr << "test passed!\n";
		}
	}
}
```

```python
import random
import subprocess

def generate():
	return random.randint(1, 100)

while True:
	test = str(generate())
	str1 = subprocess.run(["./solution"], input=test, capture_output=True, text=True)
	str2 = subprocess.run(["./stupid"], input=test, caprure_output=True, text=True)
	if (str1 != str2):
		print("Test failed!\n" + test)
```
